# 开发计划：读写分离沙箱规则 (Plan/Read/Write Separation Sandbox Rules)

> **目标**: 让 agent 自由读取工作空间外部目录（敏感文件除外），写操作仍受沙箱边界限制。
> 外部写需经 HITL 审批后方可执行。不改 HITL 链路和前端，仅修改 workspace.py 和 agents.py。
>
> **方案**: 方案 B — 精细化（ guarded 内部写直走，仅外部写 HITL）
> **核心改动**: `resolve_path()` 拆分为 `resolve_read_path()`（开放读）和 `resolve_write_path()`（限制写）

```python
    """Raised when a path violates workspace security policy."""
    REASONS = {
```

### 1.5 新增 `resolve_read_path()`
```python
def resolve_read_path(self, file_path: str) -> Path:
    """Resolve path for reading — external paths allowed.
    Guards: (1) symlink-based path traversal, (2) sensitive file detection.
    """
    candidate = self._resolve(file_path)

    # Guard 1a: relative path must not lexically escape
    is_relative = not file_path.startswith("/") and not file_path.startswith("~")
    if is_relative:
        if candidate != self.root and self.root not in candidate.parents:
            raise PathBoundaryError(file_path, "relative_escape")

    # Guard 1b: resolved path must not escape after symlink resolution
    if candidate != self.root and self.root not in candidate.parents:
        raise PathBoundaryError(file_path, "relative_escape")

    # Guard 2: sensitive files blocked regardless of path
    if self._is_sensitive_file(candidate):
        raise PathBoundaryError(file_path, "sensitive_read")

    return candidate
```

### 1.6 新增 `resolve_write_path()`
```python
def resolve_write_path(self, file_path: str) -> Path:
    """Resolve path for writing — internal paths always allowed, external only when HITL-approved.
    When _allow_external_write is True (set by resume_interrupt after HITL approval): external writes accepted.
    """
    candidate = self._resolve(file_path)
    if candidate == self.root or self.root in candidate.parents:
        return candidate  # Inside workspace — always allowed
    if self._allow_external_write:
        return candidate  # HITL approved
    raise ExternalWriteError(file_path)  # Must be HITL-approved first
```

### 1.7 新增 `_is_sensitive_file()`

```python
@staticmethod
def _is_sensitive_file(path: Path) -> bool:
    name = path.name
    if name in SENSITIVE_BASENAMES:
        return True
    name_lower = name.lower()
    return any(name_lower.endswith(s.lower()) for s in SENSITIVE_SUFFIXES)
```

```python
def _is_external_path_candidate(file_path: str, workspace_root: Path) -> bool:
    """Lightweight: resolve the path against workspace_root, check containment.
    Returns True if likely outside, False if inside or undeterminable."""
    if not file_path or not file_path.strip():
        return False
    path_str = file_path.strip()
    if path_str.startswith("/") or path_str.startswith("~"):
        try:
            resolved = Path(path_str).expanduser().resolve()
            if resolved != workspace_root and workspace_root not in resolved.parents:
                return True
        except (OSError, ValueError):
            return False
    try:
        resolved = (workspace_root / path_str).resolve()
        if resolved != workspace_root and workspace_root not in resolved.parents:
            return True
    except (OSError, ValueError):
        return False
    return False
```

```python
def needs_write_approval(req: Any) -> bool:
    state = req.state
    if normalize_phase(state.get("phase"), state.get("work_mode")) != "execute":
        return False
    autonomy = normalize_autonomy(state.get("autonomy"))
    if autonomy == "autonomous":
        return False  # autonomous → all writes direct
    if autonomy == "supervised":
        return True  # supervised → all writes HITL
    # guarded: only external writes HITL
    if autonomy == "guarded" and workspace_root is not None:
        tool_call_name = str(getattr(req, "tool_call", None).__name__ or "") if hasattr(req, "tool_call") else ""
        file_path = _path_from_tool_input(tool_call_name, str(getattr(req, "input", {}) or ""))
        if file_path and _is_external_path_candidate(file_path, workspace_root):
            return True  # external → HITL
        return False   # internal → direct
    return True  # fallback: conservative
```
